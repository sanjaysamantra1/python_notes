Program-4 : Path & Query Param
==============================
from fastapi import FastAPI
app = FastAPI()
products = [
    {"id": 1, "name": "Laptop", "category": "Electronics", "price": 75000},
    {"id": 2, "name": "Mobile", "category": "Electronics", "price": 30000},
    {"id": 3, "name": "Keyboard", "category": "Accessories", "price": 2000},
]
# Path Parameter
@app.get("/products/{product_id}")
def get_product(product_id: int):
    for product in products:
        if product["id"] == product_id:
            return product
    return {"message": "Product not found"}

# Query Parameter
@app.get("/products")
def get_products(category: str | None = None):
    if category:
        return [
            product
            for product in products
            if product["category"].lower() == category.lower()
        ]
    return products

from fastapi import FastAPI
from employee_data import employees

app = FastAPI()

@app.get("/employees")
def get_employees():
    return employees

@app.get("/employees/{employee_id}")
def get_employee(employee_id: int):
    for employee in employees:
        if employee["id"] == employee_id:
            return employee
    return {"message": "Employee not found"}

@app.post("/employees")
def create_employee(employee: dict):
    employee["id"] = max(
        [e["id"] for e in employees],
        default=0
    ) + 1
    employees.append(employee)
    return employee

@app.put("/employees/{employee_id}")
def update_employee(employee_id: int, updated_employee: dict):
    for employee in employees:
        if employee["id"] == employee_id:
            employee.update(updated_employee)
            employee["id"] = employee_id
            return employee
    return {"message": "Employee not found"}

@app.delete("/employees/{employee_id}")
def delete_employee(employee_id: int):
    for employee in employees:
        if employee["id"] == employee_id:
            employees.remove(employee)
            return {
                "message": "Employee deleted successfully"
            }
    return {"message": "Employee not found"}

=====================================
from fastapi import FastAPI
from pydantic import BaseModel, EmailStr, Field

app = FastAPI()

class UserCreate(BaseModel):
    name: str = Field(min_length=2)
    username: str = Field(min_length=3)
    email: EmailStr

class UserResponse(BaseModel):
    id: int
    name: str
    username: str
    email: EmailStr

users = []

@app.post("/users", response_model=UserResponse)
def create_user(user: UserCreate):

    new_user = {
        "id": len(users) + 1,
        "name": user.name,
        "username": user.username,
        "email": user.email
    }

    users.append(new_user)
    return new_user


@app.get("/users", response_model=list[UserResponse])
def get_users():
    return users


@app.get("/users/{user_id}", response_model=UserResponse)
def get_user(user_id: int):

    for user in users:
        if user["id"] == user_id:
            return user
    return {
        "id": 0,
        "name": "Not Found",
        "username": "",
        "email": "notfound@example.com"
    }

from fastapi import FastAPI, Cookie, Response
app = FastAPI()
@app.get("/login")
async def login(response: Response):
    # Set a cookie in the response
    response.set_cookie(
        key="session_id",
        value="abc123",
        httponly=True,
        secure=False,   # True when using HTTPS
        samesite="lax",
        max_age=3600,
    )

    return {"message": "Logged in"}


@app.get("/profile")
async def profile(
    session_id: str | None = Cookie(default=None),
):
    # Read cookie sent by the browser
    if not session_id:
        return {"error": "Not logged in"}

    return {
        "message": "Logged in",
        "session_id": session_id,
    }


@app.get("/logout")
async def logout(response: Response):
    # Delete the cookie
    response.delete_cookie("session_id")

    return {"message": "Logged out"}

UPLOAD_DIR = Path("uploads")
UPLOAD_DIR.mkdir(parents=True, exist_ok=True)

ALLOWED_TYPES = ["image/jpeg", "image/png", "image/svg"]
MAX_FILE_SIZE = 10 * 1024 * 1024  # 10 MB

@app.post("/upload")
async def upload_file(file: UploadFile):
    if file.content_type not in ALLOWED_TYPES:
        raise HTTPException(status_code=400,detail="only JPG, PNG, SVG file supported")
    # create a new file in uploads folder, copy the content from client file to server file
    server_file = UPLOAD_DIR/file.filename
    # Copy client file -> server file
    with open(server_file, "wb") as destination:
        shutil.copyfileobj(file.file, destination)

    return {
        "message": "File copied successfully",
        "original_file": file.filename,
        "server_file": str(server_file)
    }
